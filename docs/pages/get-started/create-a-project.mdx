import React, { useState, useEffect } from 'react';
import { View, Text, TouchableOpacity, FlatList, Image, StyleSheet } from 'react-native';
import { Audio } from 'expo-av';

const characters = [
  { name: 'Ankaralı Yavuz', image: require('./assets/chars/yavuz.png') },
  { name: 'Dilber', image: require('./assets/chars/dilber.png') },
  { name: 'Kostak Hacelim', image: require('./assets/chars/kostak.png') },
  { name: 'Zeyno', image: require('./assets/chars/zeyno.png') },
  { name: 'Halaycı Memo', image: require('./assets/chars/memo.png') },
  { name: 'Kına Gülü', image: require('./assets/chars/kina.png') },
  { name: 'Oyunbaz Fato', image: require('./assets/chars/fato.png') },
  { name: 'Bağlamacı Kadir', image: require('./assets/chars/kadir.png') },
  { name: 'Efe', image: require('./assets/chars/efe.png') },
  { name: 'Ankaralı Sultan', image: require('./assets/chars/sultan.png') },
  // 10 karakter daha aynı formatta eklenebilir
];

const pavyonScenes = [
  require('./assets/stage/pavyon1.png'),
  require('./assets/stage/pavyon2.png'),
  require('./assets/stage/pavyon3.png'),
  require('./assets/stage/pavyon4.png'),
  // 250'ye kadar devam eder
];

const musicTracks = [
  require('./assets/music/ankara_baglari.mp3'),
  require('./assets/music/kostak_hacelim.mp3'),
  require('./assets/music/dilber.mp3'),
  require('./assets/music/erik_dali.mp3'),
  require('./assets/music/huriyem.mp3'),
  require('./assets/music/kesik_cayir.mp3'),
  require('./assets/music/develi.mp3'),
  require('./assets/music/la_bize_her_yer_angara.mp3'),
  // 12 şarkı daha eklenebilir
];

export default function App() {
  const [screen, setScreen] = useState('start');
  const [selectedCharacter, setSelectedCharacter] = useState(null);
  const [score, setScore] = useState(0);
  const [sound, setSound] = useState(null);
  const [sceneIndex, setSceneIndex] = useState(0);
  const [musicIndex, setMusicIndex] = useState(0);

  const startGame = async () => {
    setScreen('game');
    const { sound } = await Audio.Sound.createAsync(musicTracks[musicIndex]);
    setSound(sound);
    await sound.playAsync();
  };

  const stopGame = async () => {
    if (sound) {
      await sound.stopAsync();
    }
    setScreen('start');
    setSelectedCharacter(null);
    setScore(0);
    setSceneIndex(0);
    setMusicIndex(0);
  };

  const nextStage = () => {
    setSceneIndex((sceneIndex + 1) % pavyonScenes.length);
    setMusicIndex((musicIndex + 1) % musicTracks.length);
  };

  const increaseScore = () => {
    setScore(score + 10);
    if ((score + 10) % 50 === 0) nextStage();
  };

  if (screen === 'start') {
    return (
      <View style={styles.container}>
        <Image source={require('./assets/splash.png')} style={styles.logo} />
        <Text style={styles.title}>Ankara Pavyon Star</Text>
        <TouchableOpacity onPress={() => setScreen('select')} style={styles.button}>
          <Text style={styles.buttonText}>Oyuna Başla</Text>
        </TouchableOpacity>
      </View>
    );
  }

  if (screen === 'select') {
    return (
      <FlatList
        data={characters}
        keyExtractor={(item) => item.name}
        numColumns={2}
        renderItem={({ item }) => (
          <TouchableOpacity
            onPress={() => {
              setSelectedCharacter(item);
              startGame();
            }}
            style={styles.card}
          >
            <Image source={item.image} style={styles.image} />
            <Text style={styles.name}>{item.name}</Text>
          </TouchableOpacity>
        )}
        contentContainerStyle={styles.listContainer}
      />
    );
  }

  if (screen === 'game' && selectedCharacter) {
    return (
      <View style={styles.container}>
        <Image source={pavyonScenes[sceneIndex]} style={styles.bg} />
        <Text style={styles.score}>Puan: {score}</Text>
        <Image source={selectedCharacter.image} style={styles.character} />
        <TouchableOpacity onPress={increaseScore} style={styles.button}>
          <Text style={styles.buttonText}>Dans Et</Text>
        </TouchableOpacity>
        <TouchableOpacity onPress={stopGame} style={[styles.button, { backgroundColor: '#444' }]}>
          <Text style={styles.buttonText}>Oyunu Durdur</Text>
        </TouchableOpacity>
      </View>
    );
  }

  return null;
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#000', alignItems: 'center', justifyContent: 'center' },
  title: { fontSize: 32, color: '#FFD700', fontWeight: 'bold', marginBottom: 20 },
  logo: { width: 200, height: 200, marginBottom: 20 },
  button: { backgroundColor: '#ff1493', padding: 15, borderRadius: 10, marginTop: 10 },
  buttonText: { color: '#fff', fontSize: 18 },
  card: {
    flex: 1, alignItems: 'center', margin: 10,
    backgroundColor: '#222', padding: 10, borderRadius: 12
  },
  image: { width: 100, height: 100 },
  name: { color: '#fff', marginTop: 10 },
  listContainer: { padding: 10 },
  bg: { position: 'absolute', width: '100%', height: '100%' },
  character: { width: 150, height: 150, marginTop: 100 },
  score: { fontSize: 20, color: '#fff', textAlign: 'center', marginTop: 10 },
});
